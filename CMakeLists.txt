cmake_minimum_required(VERSION 3.20)
project(PinkTromboneSC VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SC_PATH "${CMAKE_CURRENT_SOURCE_DIR}/vendor/supercollider" CACHE PATH "SuperCollider source headers")
option(BUILD_SUPERNOVA "Build the supernova plugin too" ON)
option(BUILD_TESTING "Build standalone DSP tests" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(SANITIZERS "address,undefined" CACHE STRING "Clang sanitizer list")
if(NOT EXISTS "${SC_PATH}/include/plugin_interface/SC_PlugIn.h")
    message(FATAL_ERROR "SC_PATH must contain include/plugin_interface/SC_PlugIn.h")
endif()
if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=${SANITIZERS} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${SANITIZERS})
endif()
add_library(pink_dsp STATIC src/PinkTromboneDSP.cpp)
set_target_properties(pink_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(pink_dsp PUBLIC src)
target_compile_options(pink_dsp PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -ffp-contract=off)

function(add_pink_plugin target nova)
    add_library(${target} MODULE src/PinkTrombone.cpp)
    target_link_libraries(${target} PRIVATE pink_dsp)
    target_include_directories(${target} PRIVATE
        "${SC_PATH}/include/plugin_interface" "${SC_PATH}/include/common" "${SC_PATH}/include/server")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -ffp-contract=off)
    if(${nova})
        target_compile_definitions(${target} PRIVATE SUPERNOVA)
    endif()
    set_target_properties(${target} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
    if(APPLE)
        set_target_properties(${target} PROPERTIES SUFFIX ".scx")
    endif()
    install(TARGETS ${target} LIBRARY DESTINATION PinkTrombone)
endfunction()
add_pink_plugin(PinkTrombone FALSE)
if(BUILD_SUPERNOVA)
    add_pink_plugin(PinkTrombone_supernova TRUE)
endif()
install(DIRECTORY Classes HelpSource examples DESTINATION PinkTrombone)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION PinkTrombone)

if(BUILD_TESTING)
    enable_testing()
    add_executable(dsp_tests tests/dsp_tests.cpp)
    target_link_libraries(dsp_tests PRIVATE pink_dsp)
    add_test(NAME dsp_tests COMMAND dsp_tests)
    set_tests_properties(dsp_tests PROPERTIES TIMEOUT 60)
    add_executable(reference_render tests/reference_render.cpp)
    target_link_libraries(reference_render PRIVATE pink_dsp)
endif()
